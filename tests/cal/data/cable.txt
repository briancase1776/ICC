#!/bin/bash
# A: capacity of one lane of a seat's cable DEPTH pipes deep, nobody reading.
# mesh 3, LANES 2, DEPTH D. Cable 0->1 and cable 0->2 are walked from the
# tail the seat holds back to the head (the outlet of seat 0's tee).
#  mode a: one dd per 4096 block into the head's lane 0, timeout 1 on each;
#          the first that times out is the one that waited.
#  mode b: one dd streaming 4096 blocks into cable 0->2's head, cut off by
#          SIGINT after 3 s; its records out is what went in.
#  mode c: one dd per block into seat 1's write end: the write pipe, seat 1's
#          two-outlet tee, then its cables to 0 and 2.
# Every mode then drains the tail and checks the bytes are the ones written.
S=/home/user/ICC/.claude/skills
W=$1; nDepth=$2; iTrial=$3
T=$(mktemp -d "$W/a.XXXXXX")
pPatch=
trap '[ -n "$pPatch" ] && $S/icc-patch/scripts/remove "$pPatch"; rm -rf "$T"' EXIT
t0=$SECONDS
pPatch=$($S/icc-patch/scripts/create mesh 3 2 "$nDepth")
tMake=$((SECONDS - t0))
$S/icc-patch/scripts/list | grep -qx "$pPatch up mesh 3 2 $nDepth" || { echo "patch not up"; exit 1; }
# one-outlet tees of this patch: DST -> SRC
declare -A hSrcOf=()
while read -r pPiece pScripts; do
  case $pPiece in /tmp/icc-tee-*) ;; *) continue;; esac
  mapfile -t a < "$pPiece/tee"
  [ "${#a[@]}" -eq 3 ] && hSrcOf[${a[2]}]=${a[0]}
done < "$pPatch/made"
fChain() { # writer reader -> chain head..tail on stdout
  local w=$1 r=$2 p
  p=$(awk -v w="$w" -v r="$r" 'NR>1 && $1==r && $2==1 && $4==w {print $3}' "$pPatch/patch")
  local a=("$p")
  while [ -n "${hSrcOf[$p]:-}" ]; do p=${hSrcOf[$p]}; a=("$p" "${a[@]}"); done
  printf '%s\n' "${a[@]}"
}
head -c $((700*4096)) /dev/urandom > "$T/src"
fDrain() { # tail nBlocks [outfile]
  local o=${3:-$T/out}
  timeout 3 dd if="$1/0" bs=4096 status=none > "$o"
  local nGot; nGot=$(wc -c < "$o")
  if [ "$nGot" -eq $(( $2 * 4096 )) ] && cmp -s "$o" <(head -c "$nGot" "$T/src"); then echo intact; else echo "DRAINED $nGot WANT $(( $2 * 4096 ))"; fi
}
mapfile -t aC1 < <(fChain 0 1)
mapfile -t aC2 < <(fChain 0 2)
[ "${#aC1[@]}" -eq "$nDepth" ] && [ "${#aC2[@]}" -eq "$nDepth" ] || { echo "chain length ${#aC1[@]}/${#aC2[@]} != $nDepth"; exit 1; }
# mode a
n=0
while timeout 1 dd if="$T/src" of="${aC1[0]}/0" bs=4096 skip=$n count=1 status=none; do n=$((n+1)); done
nA=$n; sA=$(fDrain "${aC1[-1]}" "$nA")
# mode b
timeout -s INT 3 dd if="$T/src" of="${aC2[0]}/0" bs=4096 2> "$T/err"
nB=$(grep 'records out' "$T/err" | cut -d+ -f1); pB=$(grep 'records out' "$T/err" | cut -d+ -f2 | cut -d' ' -f1)
sB=$(fDrain "${aC2[-1]}" "$nB")
# mode c: seat 1's write end
pW1=$(awk 'NR>1 && $1==1 && $2==0 {print $3}' "$pPatch/patch")
n=0
while timeout 1 dd if="$T/src" of="$pW1/0" bs=4096 skip=$n count=1 status=none; do n=$((n+1)); done
nC=$n
mapfile -t aC10 < <(fChain 1 0); mapfile -t aC12 < <(fChain 1 2)
fDrain "${aC10[-1]}" "$nC" "$T/o0" > "$T/s0" & fDrain "${aC12[-1]}" "$nC" "$T/o2" > "$T/s2" & wait
sC0=$(cat "$T/s0"); sC2=$(cat "$T/s2")
echo "D=$nDepth trial=$iTrial made_in=${tMake}s 17D-1=$((17*nDepth-1)) | a(dd per block, cable 0->1)=$nA [$sA] | b(one dd streaming, cable 0->2)=$nB full +$pB partial [$sB] | c(dd per block, seat 1 write end)=$nC [to0 $sC0, to2 $sC2] 17(D+1)-1=$((17*nDepth+16))"
