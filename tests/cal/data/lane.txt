#!/bin/bash
# B2: printf of N bytes | dd bs=4096 [iflag=fullblock] of=LANE, a reader
# (head -c N) draining the lane. Per run: did dd finish (rc), did the reader
# get the N bytes intact, dd's records out, and (under strace) every write
# size dd made on the lane.
S=/home/user/ICC/.claude/skills
W=$1; nT=$2; eStrace=$3; osFlag=$4; shift 4
P=$($S/icc-pipes/scripts/create 2)
trap '$S/icc-pipes/scripts/remove "$P"' EXIT
[ -f "$W/txt" ] || head -c 240000 /dev/urandom | base64 -w0 | head -c 300000 > "$W/txt"
osAll=$(cat "$W/txt")
for nSize in "$@"; do
  osBytes=${osAll:0:nSize}
  printf '%s' "$osBytes" > "$W/want"
  nFin=0; nIntact=0; declare -A hOut=(); declare -A hShape=()
  for ((i=0;i<nT;i++)); do
    timeout 20 head -c "$nSize" "$P/0" > "$W/got" & pidR=$!
    if [ "$eStrace" = y ]; then
      printf '%s' "$osBytes" | timeout 15 strace -o "$W/b2.st" -e trace=write dd bs=4096 $osFlag of="$P/0" 2> "$W/b2.err"; rc=${PIPESTATUS[1]}
    else
      printf '%s' "$osBytes" | timeout 15 dd bs=4096 $osFlag of="$P/0" 2> "$W/b2.err"; rc=${PIPESTATUS[1]}
    fi
    wait $pidR
    [ "$rc" -eq 0 ] && nFin=$((nFin+1))
    cmp -s "$W/got" "$W/want" && nIntact=$((nIntact+1))
    osOut=$(grep 'records out' "$W/b2.err" | cut -d' ' -f1)
    hOut[$osOut]=$(( ${hOut[$osOut]:-0} + 1 ))
    if [ "$eStrace" = y ]; then
      # shape: sizes of writes to fd 1 (the lane), in order, run-length coded
      osShape=$(grep '^write(1,' "$W/b2.st" | sed -E 's/.*= ([0-9-]+)$/\1/' | uniq -c | awk '{printf "%sx%s ", $1, $2}')
      hShape[$osShape]=$(( ${hShape[$osShape]:-0} + 1 ))
    fi
  done
  osO=; for k in "${!hOut[@]}"; do osO="$osO $k(x${hOut[$k]})"; done
  echo "N=$nSize flag=${osFlag:-none} strace=$eStrace trials=$nT finished=$nFin intact=$nIntact records_out:$osO"
  for k in "${!hShape[@]}"; do echo "     writes: $k (x${hShape[$k]})"; done
  unset hOut hShape
done

import os, sys, errno
lane = sys.argv[1]
def fill(pattern, pre=(), reads=()):
    """write pre writes, then repeat pattern until a write would block; return bytes held"""
    fd = os.open(lane, os.O_WRONLY | os.O_NONBLOCK)
    rfd = os.open(lane, os.O_RDONLY | os.O_NONBLOCK)
    held = 0; nwrites = 0
    try:
        for step in pre:
            if step[0] == 'w':
                held += os.write(fd, b'x' * step[1]); nwrites += 1
            else:
                held -= len(os.read(rfd, step[1]))
        i = 0
        while True:
            s = pattern[i % len(pattern)]
            try:
                n = os.write(fd, b'x' * s)
            except BlockingIOError:
                break
            if n != s:  # partial (only possible for s > PIPE_BUF)
                held += n; nwrites += 1; break
            held += n; nwrites += 1; i += 1
    finally:
        # drain and verify
        got = 0
        while True:
            try:
                b = os.read(rfd, 65536)
            except BlockingIOError:
                break
            if not b: break
            got += len(b)
        os.close(fd); os.close(rfd)
    assert got == held, (got, held)
    return held, nwrites

#!/bin/bash
# C: read -r -u FD for the count line, then head -c N <&FD for the payload.
# Two messages back to back on one lane; reader 1 takes message 1, a separate
# reader 2 takes message 2. Any overread by reader 1 shows as a bad message 2.
S=/home/user/ICC/.claude/skills
W=$1; nT=$2; shift 2
P=$($S/icc-pipes/scripts/create 2)
trap '$S/icc-pipes/scripts/remove "$P"' EXIT
head -c 300000 /dev/urandom > "$W/rnd"
for nSize in "$@"; do
  head -c "$nSize" "$W/rnd" > "$W/m1"
  tail -c "$nSize" "$W/rnd" > "$W/m2"
  nOk=0; nBad=0
  for ((i=0;i<nT;i++)); do
    # writer: both messages, back to back, as dd bs=4096 passes them on
    ( { printf '%s\n' "$nSize"; cat "$W/m1"; printf '%s\n' "$nSize"; cat "$W/m2"; } |
        timeout 10 dd of="$P/0" bs=4096 status=none ) &
    pidW=$!
    # reader 1
    timeout 10 bash -c '
      exec {fd}< "$1/0"
      read -r -u "$fd" n || exit 3
      [ "$n" = "$2" ] || exit 4
      head -c "$n" <&"$fd" > "$3/o1"' _ "$P" "$nSize" "$W"; rc1=$?
    # reader 2, a fresh process
    timeout 10 bash -c '
      exec {fd}< "$1/0"
      read -r -u "$fd" n || exit 3
      [ "$n" = "$2" ] || { echo "reader2 got count [$n]" >&2; exit 4; }
      head -c "$n" <&"$fd" > "$3/o2"' _ "$P" "$nSize" "$W"; rc2=$?
    wait $pidW; rcW=$?
    # nothing may be left on the lane
    nLeft=$(timeout 0.2 dd if="$P/0" bs=65536 status=none | wc -c)
    if [ $rc1 -eq 0 ] && [ $rc2 -eq 0 ] && [ $rcW -eq 0 ] && cmp -s "$W/o1" "$W/m1" && cmp -s "$W/o2" "$W/m2" && [ "$nLeft" -eq 0 ]; then
      nOk=$((nOk+1))
    else
      nBad=$((nBad+1)); echo "BAD size=$nSize i=$i rc1=$rc1 rc2=$rc2 rcW=$rcW left=$nLeft" >&2
    fi
  done
  echo "size=$nSize trials=$nT ok=$nOk bad=$nBad"
done
